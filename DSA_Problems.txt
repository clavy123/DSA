//binary search method
public class Main
{
	public static void main(String[] args) {
		int arr[] = new int[]{
    2, 5, 8, 11, 14, 17, 21, 24, 27, 31,
    34, 36, 39, 42, 45, 47, 51, 54, 57, 61,
    64, 67, 70, 73, 76, 79, 82, 85, 88, 91,
    94, 97, 101, 104, 108, 111, 115, 119, 123, 127,
    131, 136, 141, 146, 151, 157, 163, 170, 178, 189
};
		int left = 0;
		int right = arr.length -1;
		int target = 189;
		
		while(left <= right){
		    int mid = (left + right)/2;
		    if(arr[mid] == target){
		        System.out.println(arr[mid]);
		        break;
		    }
		    else if(arr[mid] < target){
		        left = mid + 1;
		    }
		    else{
		        right = mid - 1;
		    }
		}
	}
}

class Solution {
    public int maxSubArray(int[] nums) {
        int n = nums.length;
        //currentSum is for current element tracking for max
        int currentSum = nums[0];
        //max sum is telling overall highest element or max element
        int maxSum = currentSum;
        for(int i=1;i<n;i++){
            currentSum = Math.max(nums[i],nums[i]+currentSum);
            maxSum = Math.max(currentSum,maxSum);
        }
        return maxSum;
    }
}

class Solution {
    public int removeDuplicates(int[] nums) {
        int left = 0;
        int right = 0 ;
        while(right < nums.length){
            if(nums[left] != nums[right]){
                left++;
                nums[left] = nums[right];
            }
            right++;
        }
        return left + 1;
    }
}
